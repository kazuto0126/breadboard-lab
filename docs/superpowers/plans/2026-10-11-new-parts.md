## Global Constraints

- 腳位照規格 2.1 的表格（原廠資料表）；腳名以 `/` 開頭 = 低電位有效。
- 74LS73 / 76 是下降緣觸發；74LS47 輸出低電位有效、開集極（亮 = 0、不亮 = Z）；74LS47 字形照規格 2.2 的表。
- NE555：頻率是元件屬性 `frequency`（預設 1 Hz），/RESET 為 0 時 OUT = 0。
- 7 段顯示器只做共陽極，id `7seg-ca`；電容 id `capacitor-ceramic`、`capacitor-electrolytic`，屬性 `capacitance`（法拉）。
- 子專案 8 只搬類別 `ic`：NE555 會被搬、7 段顯示器與電容不會。
- `src/core` 不能 import `src/editor`；文件、介面文字、註解繁中；commit 英文；每個 commit 都 push。

function jkffModel(spec: JkffLogic): LogicModel {
  const clocks = [...new Set(spec.flipflops.map((f) => f.clock))];
  return {
    outputs: spec.flipflops.flatMap((f) => [f.q, f.qn].filter((p): p is string => p !== undefined)),
    init: () => ({ prev: {}, bits: zeros(spec.flipflops.length) }),
    evaluate(read, state) {
      const outputs: Record<string, Level> = {};
      const bits = spec.flipflops.map((ff, i) => {
        let q = state.bits[i] ?? 0;
        const pre = ff.preset ? asserted(read, ff.preset) : 0;
        const clr = ff.clear ? asserted(read, ff.clear) : 0;
        // 和 74LS74 相同：/PRE 與 /CLR 同時有效時 Q 與 /Q 都是 HIGH。
        const both = pre === 1 && clr === 1;
        if (both || pre === 1) q = 1;
        else if (clr === 1) q = 0;
        else if (pre === 'X' || clr === 'X') q = 'X';
        else if (falling(state.prev[ff.clock], read(ff.clock))) {
          const j = read(ff.j);
          const k = read(ff.k);
          if (j === 'X' || k === 'X') q = 'X';
          else if (j === 1 && k === 1) q = not(q);
          else if (j === 1) q = 1;
          else if (k === 1) q = 0;
        }
        if (ff.q) outputs[ff.q] = q;
        if (ff.qn) outputs[ff.qn] = both ? 1 : not(q);
        return q;
      });
      return { outputs, state: { prev: snapshot(read, clocks), bits } };
    },
  };
}

function comparatorModel(spec: ComparatorLogic): LogicModel {
  const o = spec.outputs;
  const c = spec.cascade;
  return {
    outputs: [o.gt, o.eq, o.lt],
    init: () => ({ prev: {}, bits: [] }),
    evaluate(read, state) {
      const a = toNumber(readAll(read, spec.a));
      const b = toNumber(readAll(read, spec.b));
      let gt: Logic;
      let eq: Logic;
      let lt: Logic;
      if (a === 'X' || b === 'X') gt = eq = lt = 'X';
      else if (a > b) [gt, eq, lt] = [1, 0, 0];
      else if (a < b) [gt, eq, lt] = [0, 0, 1];
      else {
        // 資料表的串接真值表。
        const ig = read(c.gt);
        const ie = read(c.eq);
        const il = read(c.lt);
        eq = ie;
        gt = and([not(ie), not(il)]);
        lt = and([not(ie), not(ig)]);
      }
      return { outputs: { [o.gt]: gt, [o.eq]: eq, [o.lt]: lt }, state };
    },
  };
}

function muxModel(spec: MuxLogic): LogicModel {
  return {
    outputs: spec.units.flatMap((u) => [u.output, u.outputN].filter((p): p is string => p !== undefined)),
    init: () => ({ prev: {}, bits: [] }),
    evaluate(read, state) {
      const outputs: Record<string, Level> = {};
      for (const u of spec.units) {
        const enabled = allAsserted(read, u.enable);
        const selected = toNumber(readAll(read, u.select));
        const y: Logic = enabled === 0 ? 0 : enabled === 'X' || selected === 'X' ? 'X' : read(u.data[selected]);
        if (u.output) outputs[u.output] = y;
        if (u.outputN) outputs[u.outputN] = not(y);
      }
      return { outputs, state };
    },
  };
}

/** 74LS47 的字形（亮的段 a–g），索引是輸入值 0–15（資料表：6 上面沒有橫、9 下面沒有橫）。 */
export const BCD_GLYPHS: readonly string[] = [
  'abcdef', 'bc', 'abdeg', 'abcdg', 'bcfg', 'acdfg', 'cdefg', 'abc', 'abcdefg', 'abcfg', 'deg', 'cdg', 'bfg', 'adfg', 'defg', '',
];
const SEGMENT_LETTERS = 'abcdefg';

function bcd7segModel(spec: Bcd7segLogic): LogicModel {
  return {
    outputs: [...spec.segments],
    init: () => ({ prev: {}, bits: [] }),
    evaluate(read, state) {
      const outputs: Record<string, Level> = {};
      // 亮 → 驅動 0；不亮 → 開集極斷開（Z）。
      const show = (on: (segment: string) => Logic) =>
        spec.segments.forEach((pin, i) => {
          const v = on(SEGMENT_LETTERS[i]);
          outputs[pin] = v === 1 ? 0 : v === 0 ? 'Z' : 'X';
        });
      const bi = read(spec.blanking);
      const lt = read(spec.lampTest);
      const rbi = read(spec.rippleBlanking);
      const value = toNumber(readAll(read, spec.inputs));
      if (bi === 0) show(() => 0);
      else if (bi === 'X') show(() => 'X');
      else if (lt === 0) show(() => 1);
      else if (lt === 'X' || value === 'X') show(() => 'X');
      else if (value === 0 && rbi === 0) show(() => 0);
      else if (value === 0 && rbi === 'X') show(() => 'X');
      else show((segment) => (BCD_GLYPHS[value].includes(segment) ? 1 : 0));
      return { outputs, state };
    },
  };
}

/** NE555 的虛擬振盪腳：編譯模擬時接到一個由模擬時間驅動的節點（見 simulate.ts）。 */
export const OSC_PIN = '~OSC';

function timer555Model(spec: Timer555Logic): LogicModel {
  return {
    outputs: [spec.output],
    init: () => ({ prev: {}, bits: [] }),
    evaluate(read, state) {
      const reset = asserted(read, spec.reset);
      const v: Logic = reset === 1 ? 0 : reset === 'X' ? 'X' : read(OSC_PIN);
      return { outputs: { [spec.output]: v }, state };
    },
  };
}

```ts
import type { ComponentDefinition, Rotation } from '../types/component';

export interface PinoutPin {
  number: number;
  name: string;
  /** 顯示用的名稱（去掉表示低電位有效的開頭 /）。 */
  label: string;
  activeLow: boolean;
  kind: 'power' | 'ground' | 'nc' | 'other';
}

export interface PinoutLayout {
  /** 兩排都是從左到右，和元件插在板上時看到的方向相同。 */
  top: PinoutPin[];
  bottom: PinoutPin[];
  notch: 'left' | 'right';
}

/** 規格 2026-10-11 new-parts 第 3 節：DIP 元件（IC、NE555、7 段顯示器）的腳位圖；指撥開關與其他封裝回傳 null。 */
export function pinoutLayout(def: ComponentDefinition, rotation: Rotation): PinoutLayout | null {
  if (def.package.type !== 'dip' || def.category === 'dip-switch') return null;
  const count = def.package.pinCount;
  const half = count / 2;
  const pin = (n: number): PinoutPin => {
    const d = def.pins.find((p) => p.number === n)!;
    const activeLow = d.name.startsWith('/') && d.name.length > 1;
    const kind = d.role === 'power' || d.role === 'anode' ? 'power' : d.role === 'ground' ? 'ground' : d.role === 'nc' ? 'nc' : 'other';
    return { number: n, name: d.name, label: activeLow ? d.name.slice(1) : d.name, activeLow, kind };
  };
  const range = (from: number, to: number) => {
    const out: number[] = [];
    for (let n = from; from <= to ? n <= to : n >= to; n += from <= to ? 1 : -1) out.push(n);
    return out;
  };
  // 旋轉 0：第 1 腳在左下，下排 1…half、上排 count…half+1；轉 180 度剛好上下左右對調。
  if (rotation === 180) return { top: range(half, 1).map(pin), bottom: range(half + 1, count).map(pin), notch: 'right' };
  return { top: range(count, half + 1).map(pin), bottom: range(1, half).map(pin), notch: 'left' };
}
```

/**
 * 74LS47 + 共陽極 7 段顯示器：SW1 的 4 個開關是 BCD 的 A–D（ON = 0、OFF = 1，輸入浮接讀成 HIGH），
 * 74LS47 的 7 個輸出各經一顆 330 Ω 電阻接到顯示器的段，CA 接 VCC。一開始 4 個開關都 ON，顯示 0。
 */
export function sevenSegmentCircuit(): Circuit {
  // 電阻直立跨過凹槽（E 列與 F 列各一隻腳）：第 k 顆在第 25 + 2k 欄。
  const segs: Array<[string, string, string]> = [
    // 段、74LS47 輸出所在的孔（上半）、顯示器的腳
    ['a', 'A18', 'A48'],
    ['b', 'A19', 'A49'],
    ['c', 'A20', 'J48'],
    ['d', 'A21', 'J46'],
    ['e', 'A22', 'J45'],
    ['f', 'A16', 'A46'],
    ['g', 'A17', 'A45'],
  ];
  const resistors = segs.map((_, k) => comp(`R${k + 1}`, 'resistor', [25 + 2 * k, 8], { rotation: 90, span: 3, props: { resistance: 330 } }));
  return circuit(
    [
      comp('V1', 'vcc', 'T+3'),
      comp('G1', 'gnd', 'B-3'),
      comp('G2', 'gnd', 'T-3'),
      comp('SW1', 'dipsw-4', 'F5', { props: { switchStates: [true, true, true, true] } }),
      comp('U1', '74LS47', 'F15'),
      ...resistors,
      comp('DS1', '7seg-ca', 'H45'),
    ],
    [
      // 開關上側 → GND（T- 軌）
      wire('s1', 'T-5', 'A5'),
      wire('s2', 'T-6', 'A6'),
      wire('s3', 'T-7', 'A7'),
      wire('s4', 'T-8', 'A8'),
      // 開關下側 → 74LS47 的 A（F21）、B（F15）、C（F16）、D（F20）
      wire('ia', 'J5', 'J21'),
      wire('ib', 'J6', 'I15'),
      wire('ic', 'J7', 'I16'),
      wire('id', 'J8', 'J20'),
      wire('vcc', 'T+15', 'B15'), // 第 16 腳 VCC（E15）
      wire('gnd', 'J22', 'B-22'), // 第 8 腳 GND
      wire('gg', 'T-61', 'B-61'), // 上下兩條 GND 軌接通
      wire('ca', 'A47', 'T+47'), // 共陽極 → VCC
      ...segs.flatMap(([name, out, seg], k) => [wire(`${name}1`, out, [25 + 2 * k, 4]), wire(`${name}2`, [25 + 2 * k, 15], seg)]),
    ],
  );
}

/**
 * NE555 當 1 Hz 時脈推動 74LS93：NE555（U2）在 F3，R1 = 1 kΩ、R2 = 68 kΩ、C1 = 10 µF 照標準無穩態接法，C2 = 0.01 µF 接 CONT；
 * OUT 接 74LS93 的 CKA，QA–QD 各經電阻點亮一顆 LED（和「74LS93 計數器」範例相同）。按「▶ 模擬」就會自己數。
 */
export function timerCounterCircuit(): Circuit {
  const base = counterCircuit();
  const keep = new Set(['g4', 'c1']);
  return circuit(
    [
      ...base.components.filter((c) => c.id !== 'SW1'),
      comp('U2', 'NE555', 'F3'),
      comp('R5', 'resistor', 'A52', { props: { resistance: 1000 } }),
      comp('R6', 'resistor', 'B56', { props: { resistance: 68000 } }),
      comp('C1', 'capacitor-electrolytic', 'C60', { props: { capacitance: 1e-5 } }),
      comp('C2', 'capacitor-ceramic', 'J50', { props: { capacitance: 1e-8 } }),
    ],
    [
      ...base.wires.filter((w) => !keep.has(w.id)),
      wire('t1', 'E52', 'B+52'), // R1 上端 → VCC
      wire('t2', 'D56', 'A4'), // DISCH（第 7 腳 E4）
      wire('t3', 'D60', 'A5'), // THRES（第 6 腳 E5）
      wire('t4', 'J4', 'E60'), // TRIG（第 2 腳 F4）接 THRES
      wire('t5', 'A62', 'T-60'), // C1 負極 → GND
      wire('t6', 'A6', 'J50'), // CONT（第 5 腳 E6）→ C2
      wire('t7', 'I52', 'B-52'), // C2 另一腳 → GND
      wire('t8', 'J3', 'B-4'), // NE555 GND（第 1 腳 F3）
      wire('t9', 'A3', 'B+4'), // NE555 VCC（第 8 腳 E3）
      wire('t10', 'J6', 'B+6'), // /RESET（第 4 腳 F6）接 VCC
      wire('t11', 'J5', 'A10'), // OUT（第 3 腳 F5）→ 74LS93 CKA（E10）
    ],
  );
}
