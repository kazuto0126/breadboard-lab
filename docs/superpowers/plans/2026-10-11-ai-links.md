# 回答裡的孔位可以點、導線解說 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** AI 回答與導線說明裡的孔位、腳位可以點（板上閃一下）；點一條線看到不用網路的「這條線在做什麼」，並可以問 AI。

**Architecture:** 兩個新的純函式（`src/editor/aiLinks.ts` 的 `linkify`、`src/editor/explain/wire.ts` 的 `explainWire`）放在 editor 層，不碰網頁；
`brief.ts` 加上導線的 AI 訊息與依板型產生的孔位規則；`useAi` 的對話從「錯誤」改成 `AiTopic`（錯誤或導線）；主 store 加 `flash` 讓板上閃圈。

**Tech Stack:** TypeScript、React、Zustand、Vitest、Playwright

**Spec:** `docs/superpowers/specs/2026-10-11-ai-links-design.md`

## Global Constraints

- 文件、介面文字、程式註解用繁體中文；commit message 用英文（`feat(editor): …`）。
- `src/editor/`、`src/core/` 不能用瀏覽器 API（`tsconfig.json` 沒有 DOM 型別）。
- 點孔位不改選取、不換右邊面板分頁、不進 Undo（規格第 4 節）。
- 只有電路裡真的存在的孔或腳才變成可以點（規格第 1 節）。
- 測試不連真的 Gemini（單元測試 `vi.mock`，e2e 用 `page.route` 攔截）。
- 每個 task 結束跑 `npm run typecheck && npm test`；改到畫面時跑 e2e（本機：`PLAYWRIGHT_CHROMIUM_EXECUTABLE="C:/Program Files (x86)/Microsoft/Edge/Application/msedge.exe"`）。

## Review Focus

1. 回答裡的元件 ID（D1、C1）和孔位同名 → 不能變成孔位連結（Task 1 測試）。
2. 多塊 MB-102 時沒有前綴的「A10」→ 不變成連結；`b2:A12` 要指到 b2（Task 1 測試）。
3. 轉接排迴圈（同一 Net 的線繞成一圈）→ `explainWire` 不能無窮迴圈（Task 2 測試）。
4. 串流中途的半截孔位（「A1」之後才收到「0」）→ 每次重新辨識整段文字，最後結果正確（Task 6 用 `LinkedText` 每次渲染重算，不快取片段）。
5. 換成導線對話後，錯誤卡上的舊對話不能出現在導線卡上，反之亦然（Task 4 測試）。

```ts
import { parseHoleName } from '../core/breadboard/naming';
import { getBoard, getLayout } from '../core/breadboard/registry';
import { resolveFootprint } from '../core/circuit/footprint';
import { getComponentDefinition } from '../core/circuit/library';
import type { Circuit } from '../types/circuit';
import type { GridPoint } from '../types/geometry';

/** 文字裡可以點的一段：點了在板上閃 points。 */
export interface AiLink {
  text: string;
  points: GridPoint[];
}

// 規格第 2 節：腳位（U1 第 3 腳、U1 的第 3 腳、U1 pin 3）、接線柱（紅柱、紅色接線柱）、孔位（A10、T+10、P3+12、2-F30、b2:A12）。
const PIN = /(?<![A-Za-z0-9])([A-Za-z][A-Za-z0-9]*)(?:\s*的)?\s*第\s*(\d+)\s*腳|(?<![A-Za-z0-9])([A-Za-z][A-Za-z0-9]*)\s+[Pp][Ii][Nn]\s*(\d+)/g;
const POST = /(紅|黑|黃|綠)(?:色接線柱|柱)/g;
const HOLE = /(?<![A-Za-z0-9+\-−–:])(?:([A-Za-z]+\d*):)?((?:[TB][+\-−–]|P[1-5][+\-−–]|[1-3]-[A-J]|[A-J])\d+)(?![A-Za-z0-9+\-−–])/g;

function pinPointOf(circuit: Circuit, id: string, pinNumber: number): GridPoint | null {
  const instance = circuit.components.find((c) => c.id === id);
  const def = instance ? getComponentDefinition(instance.defId) : undefined;
  return (instance && def && resolveFootprint(instance, def).pins.find((p) => p.pinNumber === pinNumber)?.point) || null;
}

/** 接線柱：依 circuit.boards 順序第一塊有這支柱子的板。 */
function postPoint(circuit: Circuit, name: string): GridPoint | null {
  for (const b of circuit.boards) {
    const post = getLayout(b.layoutId).posts?.find((p) => p.name === name);
    if (post) return { boardId: b.id, x: post.x, y: post.y };
  }
  return null;
}

/** 有前綴只找那塊板；沒有前綴時只有一塊板有這個孔名才算，和元件 ID 同名的不算。 */
function holePoint(circuit: Circuit, prefix: string | undefined, name: string): GridPoint | null {
  if (prefix) {
    const b = circuit.boards.find((x) => x.id === prefix);
    return b ? parseHoleName(getBoard(b.layoutId, b.id), name) : null;
  }
  if (circuit.components.some((c) => c.id === name)) return null;
  const hits = circuit.boards.map((b) => parseHoleName(getBoard(b.layoutId, b.id), name)).filter((p): p is GridPoint => p !== null);
  return hits.length === 1 ? hits[0] : null;
}

interface Hit {
  start: number;
  end: number;
  priority: number;
  points: GridPoint[];
}

/** 規格第 2 節：把文字切成一般文字與可以點的片段；只有電路裡真的存在的孔或腳才變成連結。 */
export function linkify(circuit: Circuit, text: string): Array<string | AiLink> {
  const hits: Hit[] = [];
  const scan = (re: RegExp, priority: number, resolve: (m: RegExpExecArray) => GridPoint | null) => {
    for (const m of text.matchAll(re)) {
      const p = resolve(m);
      if (p) hits.push({ start: m.index!, end: m.index! + m[0].length, priority, points: [p] });
    }
  };
  scan(PIN, 0, (m) => pinPointOf(circuit, m[1] ?? m[3], Number(m[2] ?? m[4])));
  scan(POST, 1, (m) => postPoint(circuit, m[1]));
  scan(HOLE, 2, (m) => holePoint(circuit, m[1], m[2]));

  const taken: Hit[] = [];
  for (const h of hits.sort((a, b) => a.priority - b.priority || a.start - b.start)) {
    if (taken.every((t) => h.end <= t.start || h.start >= t.end)) taken.push(h);
  }
  taken.sort((a, b) => a.start - b.start);
  const out: Array<string | AiLink> = [];
  let at = 0;
  for (const h of taken) {
    if (h.start > at) out.push(text.slice(at, h.start));
    out.push({ text: text.slice(h.start, h.end), points: h.points });
    at = h.end;
  }
  if (at < text.length) out.push(text.slice(at));
  return out;
}
```

```ts
import { nodeIdOf } from '../../core/breadboard/naming';
import { groupInfo, groupOfPoint, indexCircuit, type CircuitIndex } from '../../core/circuit/circuitIndex';
import { compareIds } from '../../core/circuit/nets';
import type { RailLabel } from '../../types/board';
import type { Circuit, Net, PinRef, Wire } from '../../types/circuit';
import type { GridPoint } from '../../types/geometry';
import { analyzeCircuit, holeLabel } from '../analysis';
import { straightWireProblem, type StraightProblem } from '../manualWire';
import { freeHolesInGroup, RAIL_TEXT } from './helpers';

export interface WireExplanation {
  net: Net | undefined;
  summary: string;
  chain: Array<{ wireId: string; text: string }>;
  why: string | null;
}

const NET_KIND: Record<Net['kind'], string> = { power: '電源', ground: '地', signal: '訊號' };
const POST_TEXT: Record<string, string> = { red: '紅色接線柱（+5 V）', black: '黑色接線柱（GND）', yellow: '黃色接線柱', green: '綠色接線柱' };

/** 一排（導通群組）上的東西：元件腳、電源軌、接線柱、接在這排的導線。 */
interface Row {
  id: string;
  pins: PinRef[];
  rail: RailLabel | null;
  post: string | null;
  wires: Wire[];
}

function rowsOf(circuit: Circuit, index: CircuitIndex): (p: GridPoint) => Row | null {
  const cache = new Map<string, Row>();
  const wiresAt = new Map<string, Wire[]>();
  for (const w of circuit.wires) {
    for (const g of new Set([groupOfPoint(index, w.start), groupOfPoint(index, w.end)])) if (g) wiresAt.set(g, [...(wiresAt.get(g) ?? []), w]);
  }
  return (p) => {
    const id = groupOfPoint(index, p);
    if (!id) return null;
    let row = cache.get(id);
    if (!row) {
      const info = groupInfo(index, id);
      const local = id.slice(id.indexOf(':') + 1);
      const pins = (info?.nodeIds ?? []).flatMap((n) => index.pinsAt.get(n) ?? []);
      pins.sort((a, b) => compareIds(a.componentId, b.componentId) || a.pinNumber - b.pinNumber);
      row = {
        id,
        pins,
        rail: local.startsWith('rail-') ? (local.slice(5) as RailLabel) : null,
        post: local.startsWith('post-') ? local.slice(5) : null,
        wires: wiresAt.get(id) ?? [],
      };
      cache.set(id, row);
    }
    return row;
  };
}

function pinText(index: CircuitIndex, ref: PinRef): string {
  const name = index.placedById.get(ref.componentId)?.def.pins.find((p) => p.number === ref.pinNumber)?.name;
  return `${ref.componentId} 第 ${ref.pinNumber} 腳${name ? `（${name}）` : ''}`;
}

/** 轉接排：沒有腳、不是電源軌或接線柱、剛好接兩條線。 */
const isHop = (row: Row) => row.pins.length === 0 && !row.rail && !row.post && row.wires.length === 2;
const connected = (row: Row) => row.pins.length > 0 || row.rail !== null || row.post !== null;

/** 一段連線的一端怎麼說（規格 3.1 的表）。 */
function endText(circuit: Circuit, index: CircuitIndex, row: Row, at: GridPoint): string {
  if (row.pins.length > 0) return row.pins.map((r) => pinText(index, r)).join('、');
  if (row.rail) return RAIL_TEXT[row.rail];
  if (row.post) return POST_TEXT[row.post] ?? '接線柱';
  if (row.wires.length >= 3) return `${holeLabel(circuit, at)} 那一排（從這裡分成 ${row.wires.length} 條線）`;
  return `${holeLabel(circuit, at)} 那一排（沒有接到別的東西）`;
}

/** 一條線的某一端那一排是什麼（chain 每一行用）。 */
function endNote(circuit: Circuit, index: CircuitIndex, row: Row, at: GridPoint): string {
  const label = holeLabel(circuit, at);
  if (row.pins.length > 0) return `${label} 和 ${row.pins.map((r) => pinText(index, r)).join('、')}同一排`;
  if (row.rail) return `${label} 插在${RAIL_TEXT[row.rail]}`;
  if (row.post) return `${label} 是${POST_TEXT[row.post] ?? '接線柱'}`;
  if (isHop(row)) return `${label} 那一排是空的，當成轉接排`;
  if (row.wires.length >= 3) return `${label} 那一排是分岔點（接 ${row.wires.length} 條線）`;
  return `${label} 那一排沒有接到別的東西`;
}

function problemText(circuit: Circuit, index: CircuitIndex, problem: StraightProblem): string {
  switch (problem.kind) {
    case 'obstacle':
    case 'pin': {
      const id = nodeIdOf(problem.point);
      const label = holeLabel(circuit, problem.point);
      const pin = index.pinsAt.get(id)?.[0];
      if (pin) return `會經過 ${pin.componentId} 的腳（${label}）`;
      const body = index.bodiesAt.get(id)?.[0];
      if (body) return `會經過 ${body} 的本體（${label}）`;
      return `會經過別條線的端點（${label}）`;
    }
    case 'overlap':
      return `會和導線 ${problem.wireId} 重疊`;
    case 'covered':
      return `端點 ${holeLabel(circuit, problem.point)} 上有導線 ${problem.wireId} 經過`;
    default:
      return '不能直接接';
  }
}

/** 規格 3.2：拿掉這段連線後，兩端能不能用一條直線接。 */
function whyText(circuit: Circuit, index: CircuitIndex, chainIds: Set<string>, a: GridPoint, b: GridPoint): string {
  const without: Circuit = { ...circuit, wires: circuit.wires.filter((w) => !chainIds.has(w.id)) };
  const pairs: Array<[GridPoint, GridPoint]> = [];
  for (const p of freeHolesInGroup(without, a)) {
    for (const q of freeHolesInGroup(without, b)) if (p.boardId === q.boardId && (p.x === q.x || p.y === q.y)) pairs.push([p, q]);
  }
  if (pairs.length === 0) return '兩端不在同一列、也不在同一欄，一條直線接不到，所以經過轉接排轉彎。';
  const name = ([p, q]: [GridPoint, GridPoint]) => `${holeLabel(circuit, p)} → ${holeLabel(circuit, q)}`;
  const ok = pairs.find(([p, q]) => straightWireProblem(without, p, q) === null);
  if (ok) return `其實也可以一條線直接接（例如 ${name(ok)}）。AI 標準佈線為了少和別的線交叉、讓出位置給其他連線，選了經過轉接排的走法。`;
  const problem = straightWireProblem(without, pairs[0][0], pairs[0][1])!;
  return `兩端雖然可以排成一直線（例如 ${name(pairs[0])}），但中間${problemText(circuit, index, problem)}，所以改走轉接排。`;
}

/** 規格第 3 節：這條線在做什麼（不用網路）。 */
export function explainWire(circuit: Circuit, wireId: string): WireExplanation | null {
  const wire = circuit.wires.find((w) => w.id === wireId);
  if (!wire) return null;
  const index = indexCircuit(circuit);
  const rowAt = rowsOf(circuit, index);

  // 往一個方向走：經過轉接排就接到下一條線，直到碰到有東西的排或分岔。
  const seen = new Set([wire.id]);
  const walk = (from: GridPoint): { wires: Wire[]; end: GridPoint } => {
    const wires: Wire[] = [];
    let at = from;
    let current = wire;
    for (;;) {
      const row = rowAt(at);
      if (!row || !isHop(row)) return { wires, end: at };
      const next = row.wires.find((w) => w.id !== current.id);
      if (!next || seen.has(next.id)) return { wires, end: at };
      seen.add(next.id);
      wires.push(next);
      at = groupOfPoint(index, next.start) === row.id ? next.end : next.start;
      current = next;
    }
  };
  const left = walk(wire.start);
  const right = walk(wire.end);
  const chainWires = [...left.wires.reverse(), wire, ...right.wires];
  const [a, b] = [left.end, right.end];
  const rowA = rowAt(a)!;
  const rowB = rowAt(b)!;

  const net = analyzeCircuit(circuit).netByWire.get(wire.id);
  const from = endText(circuit, index, rowA, a);
  const to = endText(circuit, index, rowB, b);
  const netText = net ? `${net.name}（${NET_KIND[net.kind]}）` : null;
  let summary: string;
  if (!connected(rowA) && !connected(rowB)) summary = '這條線兩端都沒有接到元件或電源。';
  else if (chainWires.length === 1) summary = `這條線把 ${from} 接到${to}${netText ? `，屬於 ${netText}` : ''}。`;
  else summary = `${netText ? `這條線屬於 ${netText}，` : '這條線'}負責把 ${from} 接到${to}。`;

  const chain = chainWires.map((w) => ({
    wireId: w.id,
    text: `${holeLabel(circuit, w.start)} → ${holeLabel(circuit, w.end)}：${endNote(circuit, index, rowAt(w.start)!, w.start)}；${endNote(circuit, index, rowAt(w.end)!, w.end)}。`,
  }));
  const why = wire.source === 'auto' && chainWires.length > 1 ? whyText(circuit, index, new Set(chainWires.map((w) => w.id)), a, b) : null;
  return { net, summary, chain, why };
}
```

```ts
export function sameTopic(a: AiTopic, b: AiTopic): boolean {
  if (a.kind === 'issue' && b.kind === 'issue') return a.issue === b.issue;
  if (a.kind === 'wire' && b.kind === 'wire') return a.wireId === b.wireId;
  return false;

/** 規格 5.3：學生正在看的導線（不用網路的說明），自動佈線的線另外附上佈線規則。 */
export function wireBrief(circuit: Circuit, wireId: string): string {
  const wire = circuit.wires.find((w) => w.id === wireId);
  const e = explainWire(circuit, wireId);
  if (!wire || !e) return `【學生正在看的導線】\n${wireId}（已經不在電路裡）`;
  const auto = wire.source === 'auto';
  const lines = [
    '【學生正在看的導線】',
    `${wire.id}（${auto ? 'AI 標準佈線' : '學生手動接的線'}）：${holeLabel(circuit, wire.start)} → ${holeLabel(circuit, wire.end)}`,
    e.summary,
  ];
  if (e.chain.length > 1) {
    lines.push(`這段連線共 ${e.chain.length} 條線：`);
    e.chain.forEach((x, i) => lines.push(`${i + 1}. ${x.text.replace('：', x.wireId === wireId ? '（本條）：' : '：')}`));
  } else lines.push(e.chain[0].text);
  if (e.why) lines.push(`為什麼不一條線直接接：${e.why}`);
  return auto ? [...lines, '', ROUTING_RULES].join('\n') : lines.join('\n');
}

export function firstMessage(circuit: Circuit, topic: AiTopic, question: string): string {
  const about = topic.kind === 'issue' ? issueBrief(circuit, topic.issue) : wireBrief(circuit, topic.wireId);
  return [circuitBrief(circuit), '', about, '', '【學生的問題】', question.trim()].join('\n');
}
```
