# EIC-1106 大型麵包板（子專案 10）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增 EIC-1106 版型（3 個插孔區、5 條電源軌、4 支接線柱，紅 = VCC、黑 = GND），能新增、存檔、佈線、模擬、檢查與解說。

**Architecture:** 版型資料（`layouts.ts`）多一份 `EIC1106_LAYOUT`；terminal 列帶區號、rail 列可以是 `P1+`…`P5-`、版型有 `posts`。`buildBoard` 依版型產生節點與群組，
接線柱是 kind `rail`、帶 `source` 的單孔群組（佈線時自動和電源軌一樣：可以當終端、不能當轉接排）。Net 種類、Check Circuit、模擬與解說把 `source` 當成電源。
畫面的板子底圖改成依版型畫。

**Tech Stack:** TypeScript、Vitest、React、Playwright。

**Spec:** `docs/superpowers/specs/2026-10-11-eic-board-design.md`

## Global Constraints

- 版型 id `EIC1106`、顯示名稱 `EIC-1106`；MB-102 的孔名、群組 ID、座標完全不變。
- 孔名：插孔區 `1-A15`、電源軌 `P2+12`（解析也接受全形減號）、接線柱 `紅`、`黑`、`黃`、`綠`。
- 座標照規格第 2 節的表（height 56，接線柱在 y = 2、x = 20 / 28 / 36 / 44）。
- 紅柱 = VCC 5 V、黑柱 = GND；黃、綠不是電源。
- `src/core` 不能 import `src/editor`；繁中文件、英文 commit；每個 commit 都 push。

## Review Focus

1. 舊存檔與舊測試（只有 MB-102）行為完全不變 → 每個 Task 結束都跑全部測試；Task 1「keeps MB-102 names and groups unchanged」。
2. 黃柱接到 IC 的 VCC 腳時不能被當成有電 → Task 2「a yellow post is not a power source」。
3. 紅柱和黑柱接在一起要報短路 → Task 2「red and black posts tied together are a short」。
4. 接線柱不能被自動佈線拿來當轉接排 → Task 5「never uses a binding post as a via」。
5. 在 EIC-1106 上沒有電時，解說要叫學生從紅柱接線，而不是找不存在的「上方的 + 電源軌」 → Task 3 測試。

```ts
const POST_COLORS: Record<string, string> = { red: '紅', black: '黑', yellow: '黃', green: '綠' };

/** 導通群組的中文說明；withBoard 為 true 時加上板子 ID（多塊板時使用）。 */
export function describeGroup(groupId: string, withBoard = false): string {
  const colon = groupId.indexOf(':');
  const local = groupId.slice(colon + 1);
  const prefix = withBoard ? `${groupId.slice(0, colon)} ` : '';
  const column = /^(?:a(\d+)-)?c(\d+)-(top|bot)$/.exec(local);
  if (column) return `${prefix}${column[1] ? `第 ${column[1]} 區` : ''}第 ${column[2]} 欄${column[3] === 'top' ? '上半（A–E）' : '下半（F–J）'}`;
  const post = /^post-(\w+)$/.exec(local);
  if (post) return `${prefix}${POST_COLORS[post[1]] ?? post[1]}色接線柱`;
  return `${prefix}${local.replace('rail-', '')} 電源軌`;
}
```

```ts
 * EIC-1106 大型麵包板：74LS00 驗收電路放在插孔區 2。紅色接線柱接 P3 +、黑色接線柱接 P4 −，不用另外放 VCC / GND 符號。
 * U1 在 2-F10：第 1–7 腳 2-F10–2-F16，第 14–8 腳 2-E10–2-E16。
 */
export function eicCircuit(): Circuit {
  const at = (name: string) => holeOn('b1', name, 'EIC1106');
  return circuit(
    [comp('U1', '74LS00', at('2-F10')), comp('R1', 'resistor', at('2-G30')), comp('D1', 'led', at('2-G40'))],
    [
      wire('vp', at('紅'), at('P3+5')), // 紅柱 → P3 +

      wire('w1', at('P3+6'), at('2-A10')), // VCC → 第 14 腳
      wire('w2', at('P4-6'), at('2-J16')), // GND → 第 7 腳

      wire('w7', at('2-J41'), at('P4-41')), // LED 陰極 → GND
    ],
    [board('b1', 0, 0, 'EIC1106')],
  );
}
```

```ts
/** Net 有沒有 VCC / GND 來源：電源元件的腳，或 EIC-1106 的紅 / 黑接線柱。 */
export function netHasSource(net: Net, index: CircuitIndex, source: 'vcc' | 'gnd'): boolean {
  const role = source === 'vcc' ? 'vcc-source' : 'gnd-source';
  return net.pins.some((r) => pinDefinition(index, r)?.role === role) || net.groupIds.some((g) => groupInfo(index, g)?.source === source);
```

```ts
export function railCarries(circuit: Circuit, boardId: string, rail: Rail, role: PinRole): boolean {
  const net = analyzeCircuit(circuit).netByGroup.get(railGroupId(boardId, rail));
  if (!net) return false;
  return net.pins.some((ref) => roleOf(circuit, ref) === role) || net.groupIds.some((g) => groupSourceOf(circuit, g) === (role === 'vcc-source' ? 'vcc' : 'gnd'));
}

function groupSourceOf(circuit: Circuit, groupId: string): 'vcc' | 'gnd' | undefined {
  const instance = circuit.boards.find((b) => b.id === groupId.slice(0, groupId.indexOf(':')));
  return instance ? getBoard(instance.layoutId, instance.id).groups[groupId]?.source : undefined;
}

/** 這塊板上某個極性的電源軌，依和 y 的距離排序。 */
function railsNear(circuit: Circuit, boardId: string, polarity: '+' | '-', y: number): Array<{ rail: Rail; y: number }> {
  const instance = circuit.boards.find((b) => b.id === boardId);
  if (!instance) return [];
  return getLayout(instance.layoutId)
    .rows.flatMap((r) => (r.kind === 'rail' && r.label.endsWith(polarity) ? [{ rail: r.label, y: r.y }] : []))
    .sort((a, b) => Math.abs(a.y - y) - Math.abs(b.y - y) || a.y - b.y);
}
```

```ts
export function chooseRail(circuit: Circuit, point: GridPoint, polarity: '+' | '-'): { rail: Rail; powered: boolean } {
  const role: PinRole = polarity === '+' ? 'vcc-source' : 'gnd-source';
  const rails = railsNear(circuit, point.boardId, polarity, point.y);
  const powered = rails.find((r) => railCarries(circuit, point.boardId, r.rail, role));
  return powered ? { rail: powered.rail, powered: true } : { rail: rails[0]?.rail ?? (polarity === '+' ? 'T+' : 'B-'), powered: false };
}

/** 接線柱的孔（EIC-1106 的紅 = vcc、黑 = gnd）；沒有時回傳 null。 */
export function postHole(circuit: Circuit, boardId: string, source: 'vcc' | 'gnd'): GridPoint | null {
  const instance = circuit.boards.find((b) => b.id === boardId);
  const post = instance ? getLayout(instance.layoutId).posts?.find((p) => p.source === source) : undefined;
  return post ? { boardId, x: post.x, y: post.y } : null;
}
```
